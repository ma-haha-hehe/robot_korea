36 inline void convertNv12ToRgb(const uint8_t *data, uint16_t width, uint16_t height,
37  std::vector<uint8_t> &out) {
38  const size_t numPixels = static_cast<size_t>(width) * height;
39  out.resize(numPixels * 3);
40  const uint8_t *yData = data;
41  const uint8_t *uvData = data + numPixels;
42 
43  // Fixed-point coefficients (scaled by 256) for BT.601 YUV-to-RGB
44  constexpr int kRv = 359; // 1.402 * 256
45  constexpr int kGu = 88; // 0.344136 * 256
46  constexpr int kGv = 183; // 0.714136 * 256
47  constexpr int kBu = 454; // 1.772 * 256
48 
49  const int w = width;
50  const int h = height;
51  const int evenW = w & ~1;
52 
53  for (int y = 0; y < h; ++y) {
54  const int yRowOff = y * w;
55  const uint8_t *uvRow = uvData + (y / 2) * w;
56  const uint8_t *yRow = yData + yRowOff;
57  uint8_t *outRow = out.data() + yRowOff * 3;
58 
59  // Process pixel pairs sharing the same UV sample
60  for (int x = 0; x < evenW; x += 2) {
61  int u = static_cast<int>(uvRow[x]) - 128;
62  int v = static_cast<int>(uvRow[x + 1]) - 128;
63 
64  int crR = kRv * v;
65  int crG = kGu * u + kGv * v;
66  int crB = kBu * u;
67 
68  // Pixel 0
69  {
70  int yv = static_cast<int>(yRow[x]) << 8;
71  uint8_t *dst = outRow + x * 3;
72  dst[0] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + crR) >> 8)));
73  dst[1] = static_cast<uint8_t>(std::max(0, std::min(255, (yv - crG) >> 8)));
74  dst[2] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + crB) >> 8)));
75  }
76  // Pixel 1
77  {
78  int yv = static_cast<int>(yRow[x + 1]) << 8;
79  uint8_t *dst = outRow + (x + 1) * 3;
80  dst[0] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + crR) >> 8)));
81  dst[1] = static_cast<uint8_t>(std::max(0, std::min(255, (yv - crG) >> 8)));
82  dst[2] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + crB) >> 8)));
83  }
84  }
85 
86  // Handle potential odd last column
87  if (evenW < w) {
88  int u = static_cast<int>(uvRow[evenW]) - 128;
89  int v = static_cast<int>(uvRow[evenW + 1]) - 128;
90  int yv = static_cast<int>(yRow[evenW]) << 8;
91  uint8_t *dst = outRow + evenW * 3;
92  dst[0] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + kRv * v) >> 8)));
93  dst[1] = static_cast<uint8_t>(std::max(0, std::min(255, (yv - kGu * u - kGv * v) >> 8)));
94  dst[2] = static_cast<uint8_t>(std::max(0, std::min(255, (yv + kBu * u) >> 8)));
95  }
96  }
97 }
