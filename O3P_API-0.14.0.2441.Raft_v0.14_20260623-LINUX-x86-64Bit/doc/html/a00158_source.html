29 inline Extrinsics convertExtrinsics(const float extrinsics[7]) {
30  Extrinsics result;
31 
32  // Extract quaternion components (x, y, z, w)
33  float qx = extrinsics[3];
34  float qy = extrinsics[4];
35  float qz = extrinsics[5];
36  float qw = extrinsics[6];
37 
38  // Extract translation (tx, ty, tz)
39  result.translation[0] = extrinsics[0];
40  result.translation[1] = extrinsics[1];
41  result.translation[2] = extrinsics[2];
42 
43  // Convert quaternion to 3x3 rotation matrix (column-major)
44  // Column 0
45  result.rotation[0] = 1.0f - 2.0f * (qy * qy + qz * qz);
46  result.rotation[1] = 2.0f * (qx * qy + qw * qz);
47  result.rotation[2] = 2.0f * (qx * qz - qw * qy);
48 
49  // Column 1
50  result.rotation[3] = 2.0f * (qx * qy - qw * qz);
51  result.rotation[4] = 1.0f - 2.0f * (qx * qx + qz * qz);
52  result.rotation[5] = 2.0f * (qy * qz + qw * qx);
53 
54  // Column 2
55  result.rotation[6] = 2.0f * (qx * qz + qw * qy);
56  result.rotation[7] = 2.0f * (qy * qz - qw * qx);
57  result.rotation[8] = 1.0f - 2.0f * (qx * qx + qy * qy);
58 
59  return result;
60 }

67 inline Extrinsics invertExtrinsics(const Extrinsics &extrinsics) {
68  Extrinsics result;
69 
70  // Transpose the rotation matrix (inverse of rotation matrix is its transpose)
71  // Original is column-major, so:
72  // Column 0 becomes Row 0
73  result.rotation[0] = extrinsics.rotation[0]; // R[0][0]
74  result.rotation[3] = extrinsics.rotation[1]; // R[1][0]
75  result.rotation[6] = extrinsics.rotation[2]; // R[2][0]
76 
77  // Column 1 becomes Row 1
78  result.rotation[1] = extrinsics.rotation[3]; // R[0][1]
79  result.rotation[4] = extrinsics.rotation[4]; // R[1][1]
80  result.rotation[7] = extrinsics.rotation[5]; // R[2][1]
81 
82  // Column 2 becomes Row 2
83  result.rotation[2] = extrinsics.rotation[6]; // R[0][2]
84  result.rotation[5] = extrinsics.rotation[7]; // R[1][2]
85  result.rotation[8] = extrinsics.rotation[8]; // R[2][2]
86 
87  // Calculate inverse translation: -R^T * t
88  result.translation[0] = -(extrinsics.rotation[0] * extrinsics.translation[0] + extrinsics.rotation[1] * extrinsics.translation[1] + extrinsics.rotation[2] * extrinsics.translation[2]);
89  result.translation[1] = -(extrinsics.rotation[3] * extrinsics.translation[0] + extrinsics.rotation[4] * extrinsics.translation[1] + extrinsics.rotation[5] * extrinsics.translation[2]);
90  result.translation[2] = -(extrinsics.rotation[6] * extrinsics.translation[0] + extrinsics.rotation[7] * extrinsics.translation[1] + extrinsics.rotation[8] * extrinsics.translation[2]);
91
