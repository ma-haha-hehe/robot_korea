121 inline std::unordered_map<MetadataType, size_t> getMetadataLayoutFromBuffer(const uint8_t *buffer, size_t len) {
122  std::unordered_map<MetadataType, size_t> layout;
123  size_t fieldOffset = 0;
124 
125  while (len) {
126  if (sizeof(MetadataField) >= len) {
127  // Error - buffer should never end with field, only values
128  layout.clear();
129  break;
130  }
131 
132  MetadataField field;
133  memcpy(&field, buffer + fieldOffset, sizeof(MetadataField));
134  fieldOffset += sizeof(MetadataField);
135  len -= sizeof(MetadataField);
136 
137  if (field.type >= sizeof(METADATA_INFO) || field.offset == 0 || METADATA_INFO[field.type].size > len) {
138  // Error - Invalid field (invalid type, invalid offset)
139  layout.clear();
140  break;
141  }
142 
143  layout[field.type] = field.offset;
144  len -= METADATA_INFO[field.type].size;
145  }
146 
147  return layout;
148 }

450 inline uint64_t getFileSize(const std::string &filename) {
451  std::ifstream inputfile;
452  inputfile.open(filename.c_str(), std::ios_base::in | std::ios_base::binary);
453  if (!inputfile.is_open()) {
454  return 0;
455  }
456 
457  // get length of file:
458  inputfile.seekg(0, std::ios::end);
459  uint64_t length = static_cast<uint64_t>(inputfile.tellg());
460  inputfile.seekg(0, std::ios::beg);
461 
462  inputfile.close();
463 
464  return length;
465 }

474 template <class T>
475 inline std::vector<T> readFileToStdVector(const std::string &filename) {
476  std::vector<T> data;
477 
478  uint64_t length = getFileSize(filename);
479 
480  if (!length) {
481  throw std::runtime_error("Failed to get file size or file is empty.");
482  }
483 
484  std::ifstream f(filename.c_str(), std::ios_base::in | std::ios_base::binary);
485  if (!f.is_open()) {
486  throw std::runtime_error("Failed to open file: " + filename);
487  }
488 
489  if (length % sizeof(T) != 0) {
490  throw std::runtime_error("File size is not a multiple of container value type size.");
491  }
492 
493  data.resize(static_cast<size_t>(length) / sizeof(T));
494  f.read((char *)&data[0], static_cast<std::streamsize>(length));
495  f.close();
496 
497  return data;
498 }
