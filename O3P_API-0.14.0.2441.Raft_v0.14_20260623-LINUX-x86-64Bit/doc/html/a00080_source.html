42  if (!context || !device || !intf) {

44  }
45 
46  if (libusb_open(device, &m_handle) != LIBUSB_SUCCESS) {
47  throw std::runtime_error("Failed to open device");
48  }
49 
50  // On Linux the kernel may have a driver attached to the interface (e.g. uvcvideo
51  // or a generic driver). Ask libusb to detach it automatically when we claim so
52  // that claim_interface succeeds. The kernel driver is re-attached on release.
53  // This call returns LIBUSB_ERROR_NOT_SUPPORTED on platforms that don't need it,
54  // which is safe to ignore.
55  libusb_set_auto_detach_kernel_driver(m_handle, 1);
56 
57  // Claim the interface
58  if (libusb_claim_interface(m_handle, intf->getNumber()) != LIBUSB_SUCCESS) {
59  libusb_close(m_handle);
60  m_handle = nullptr;
61  throw std::runtime_error("Failed to claim interface");
62  }
63 
64  m_context->startEventHandler();
65  }
66

71  libusb_close(m_handle);
72  }
73  }
