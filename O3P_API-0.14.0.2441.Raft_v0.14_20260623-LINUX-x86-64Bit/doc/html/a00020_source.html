279 template <typename T>
280 void Colorizer::buildHistogram(const T *depthData, int width, int height, float depthScale) {
281  // Clear histogram
282  std::fill(m_histogram.begin(), m_histogram.end(), 0);
283 
284  // Build histogram
285  for (int i = 0; i < width * height; ++i) {
286  T depth_val = depthData[i];
287  if (depth_val > 0) {
288  int index = static_cast<int>(depth_val * depthScale * 1000.0f); // Convert to mm
289  if (index < MAX_DEPTH_VALUES) {
290  m_histogram[index]++;
291  }
292  }
293  }
294 
295  // Build cumulative histogram
296  for (int i = 1; i < MAX_DEPTH_VALUES; ++i) {
297  m_histogram[i] += m_histogram[i - 1];
298  }
299 }

301 template <typename T>
302 void Colorizer::colorizeFixedRange(const T *depthData, std::vector<uint8_t> &rgbData, int width, int height, float depthScale) {
303  m_colorMap = m_colorMaps[static_cast<int>(m_currentScheme)];
304 
305  float minInM = static_cast<float>(m_minDepthFilter) / 1000; // convert to meters
306  float maxInM = static_cast<float>(m_maxDepthFilter) / 1000; // convert to meters
307 
308  for (int i = 0; i < width * height; ++i) {
309  T depth_val = depthData[i];
310 
311  if (depth_val > 0) {
312  float depth_meters = depth_val * depthScale;
313 
314  // Normalize depth to [0,1]
315  float normalized = (depth_meters - minInM) / (maxInM - minInM); // fixed range
316  if (normalized > 1.0f)
317  normalized = 1.0f;
318  if (normalized < 0.0f)
319  normalized = 0.0f;
320  RgbColor color = m_colorMap.getColor(normalized);
321  rgbData[i * 3 + 0] = static_cast<uint8_t>(color.r);
322  rgbData[i * 3 + 1] = static_cast<uint8_t>(color.g);
323  rgbData[i * 3 + 2] = static_cast<uint8_t>(color.b);
324  } else {
325  // Invalid depth - set to black
326  rgbData[i * 3 + 0] = 0;
327  rgbData[i * 3 + 1] = 0;
328  rgbData[i * 3 + 2] = 0;
329  }
330  }
331 }

334 void Colorizer::colorizeHistogramEqualized(const T *depthData, std::vector<uint8_t> &rgbData, int width, int height, float depthScale) {
335  m_colorMap = m_colorMaps[static_cast<int>(m_currentScheme)];
336  int total_pixels = m_histogram[MAX_DEPTH_VALUES - 1];
337 
338  for (int i = 0; i < width * height; ++i) {
339  T depth_val = depthData[i];
340 
341  if (depth_val > 0) {
342  int index = static_cast<int>(depth_val * depthScale * 1000.0f); // Convert to mm
343  if (index < MAX_DEPTH_VALUES && total_pixels > 0) {
344  float normalized = static_cast<float>(m_histogram[index]) / static_cast<float>(total_pixels);
345  RgbColor color = m_colorMap.getColor(normalized);
346  rgbData[i * 3 + 0] = static_cast<uint8_t>(color.r);
347  rgbData[i * 3 + 1] = static_cast<uint8_t>(color.g);
348  rgbData[i * 3 + 2] = static_cast<uint8_t>(color.b);
